 30class SwitchTensor:
 31    """ This class connect a tensor with a switch
 32
 33    Attributes:
 34        name (str): switch name
 35        shape_set (Set[Tuple[int,...]]): a set of tensor shapes associated with this switch
 36        ph_names (List[str]): generated from the switch name
 37        vocab_name (str): generated from the switch name
 38        var_name (str): generated from the switch name
 39        value (Any): this values is used to represent x in the special tensor atom: get(_,x)
 40
 41    """
 42    def __init__(self, sw_name: str) -> None:
 43        self.value: Optional[int] = None
 44        self.name = sw_name
 45        self.shape_set:Set[Tuple[int,...]] = set([])
 46        self.type_set:Set[str] = set([])
 47        self.ph_names = self.get_placeholder_name(sw_name)
 48        self.vocab_name = self.make_vocab_name(sw_name) # update self.value
 49        self.var_name = self.make_var_name(sw_name)
 50
 51    def enabled_placeholder(self):
 52        return len(self.ph_names) == 0
 53
 54    def add_shape(self, shape: Tuple[int,...]) -> None:
 55        self.shape_set.add(shape)
 56
 57    def add_type(self, tensor_type: str) -> None:
 58        self.type_set.add(tensor_type)
 59
 60    def get_shape(self) -> Tuple[int,...]:
 61        assert len(self.shape_set) == 1, (
 62            self.name + ": shape is not unique:" + str(self.shape_set)
 63        )
 64        return list(self.shape_set)[0]

 66    def get_type(self) -> str:
 67        assert len(self.type_set) == 1, (

 74    def get_placeholder_name(name: str) -> List[Union[str, Any]]:
 75        pattern = r"(\$placeholder[0-9]+\$)"
 76        m = re.finditer(pattern, name)
 77        names = [el.group(1) for el in m]

 80    def make_vocab_name(self, name: str) -> str:
 81        m: Optional[re.Match] = re.match(r"^tensor\(get\((.*),([0-9]*)\)\)$", name)
 82        if m:
 83            name = "tensor(" + m.group(1) + ")"
 84            self.value = int(m.group(2))
 85        pattern = r"\$(placeholder[0-9]+)\$"
 86        name_no_ph: str = re.sub(pattern, "", name)

 91        return re.sub(r"[\[\],\)\(\'$]+", "_", name)

 94class VocabSet:
 95    """ This class connect a value object with a vovabrary via placeholders
 96
 97    vocab -> placeholder ->values
 98
 99    Attributes:
100        vocab_values (Dict[str, List[Any]]): the key is a vocab name, and the value ia a list of values.
101        value_index (Dict[Tuple[str,Any],int): the key is a tuple of a vocab name and a value, and the value is an index.
102
103    """
104
105    def __init__(self) -> None:
106        # vocab name => a list of values
107        self.vocab_values: Optional[Dict[str, List[Any]]] = None
108        # vocab name, value => index
109        self.value_index: Optional[Dict[Tuple[str, int], int]] = None
110
111    def build_from_ph(self, ph_graph: 'PlaceholderGraph') -> None:
112        """ This method builds vocab_values and value_index from PlaceholderGraph.
113        """
114        vocab_ph = ph_graph.vocab_ph
115        ph_values = ph_graph.ph_values #
116        if vocab_ph is None:
117            raise ValueError(f"vocab_ph in Placeholder graph '{ph_graph}' must not be None when building from placeholders.")
118        if ph_values is None:
119            raise ValueError(f"ph_values in Placeholder graph '{ph_graph}' must not be None when building from placeholders.")
120        vocab_values: Dict[str, Set[Any]] = {}
121        for vocab_name, phs in vocab_ph.items():
122            for ph in phs:
123                if vocab_name not in vocab_values:
124                    vocab_values[vocab_name] = set()
125                vocab_values[vocab_name] |= ph_values[ph]
126        self.vocab_values = {k: list(v) for k, v in vocab_values.items()}
127        self.value_index = self._build_value_index()
128
129    def _build_value_index(self) -> Dict[Tuple[str, int], int]:
130        value_index: Dict[Tuple[str, int], int] = {}
131        if self.vocab_values is None:
132            raise ValueError(f"vocab_values in VocabSet must not be None when building value_index.")
133        for vocab_name, values in self.vocab_values.items():
134            for i, v in enumerate(sorted(values)):
135                value_index[(vocab_name, v)] = i
136        return value_index
137
138    def get_values_index(self, vocab_name: str, value: Union[int, int]) -> int:
139        key = (vocab_name, value)
140        if self.value_index is None:
141            raise ValueError(f"value_index in VocabSet must not be None when getting value_index.")
142        if key in self.value_index:
143            return self.value_index[key]
144        else:
145            return 0
146
147    def get_values(self, vocab_name: str) -> Optional[List[int]]:
148        if self.vocab_values is None:
149            raise ValueError(f"vocab_values in VocabSet must not be None when getting values.")
150        if vocab_name not in self.vocab_values:
151            return None
152        return self.vocab_values[vocab_name]

155class PlaceholderGraph:
156    """ This class build a graph related to placeholders
157
158    vocab <-- sw_info --> placeholder --> values
159
160    Attributes:
161        vocab_ph (Dict[str, Set[str]]): vocab_name => a set of nemes of placeholders
162        ph_vocab (Dict[str, Set[str]]): placeholder name => a set of the vocabs
163        ph_values (Dict[str,Set[Any]]): placeholder name => a set of values
164        vocab_shape (Dict[str, Set[Tuple[int,...]]]): vocab_name => a set of shapes
165
166    """
167
168    def __init__(self) -> None:
169        self.vocab_ph: Optional[Dict[str, Set[str]]] = None
170        self.ph_vocab: Optional[Dict[str, Set[str]]] = None
171        self.ph_values: Optional[Dict[str, Set[Any]]] = None
172        self.vocab_shape: Optional[Dict[str, Set[Tuple[int, ...]]]] = None
173
174    def _build_ph_values(self, input_data: List[InputData]) -> None:
175        ph_values:Dict[str, Set[Any]] = {}
176        for g in input_data:
177            for ph in g.placeholders:
178                if ph not in ph_values:
179                    ph_values[ph] = set()
180            placeholders = [ph for ph in g.placeholders]
181            rt = np.transpose(g.records)
182            for i, item in enumerate(rt):
183                ph_values[placeholders[i]] |= set(item)
184        self.ph_values = ph_values
185
186    def _build_vocab_ph(self, ph_values: Dict[str, Set[int]], sw_info: Dict[str, SwitchTensor]) -> None:
187        # ph_vocab/vocab_ph: ph_name <== sw_info ==> vocab_name
188        # vocab_shape: vocab_name => shape
189        ph_vocab: Dict[str, Set[str]] = {ph_name: set() for ph_name in ph_values.keys()}
190        vocab_ph: Dict[str, Set[str]] = {sw.vocab_name: set() for sw in sw_info.values()}
191        vocab_shape: Dict[str, Set[Tuple[int, ...]]] = {sw.vocab_name: set() for sw in sw_info.values()}
192        for sw_name, sw in sw_info.items():
193            ## build vocab. shape
194            if sw.vocab_name not in vocab_shape:
195                vocab_shape[sw.vocab_name] = set()
196            vocab_shape[sw.vocab_name] |= sw.shape_set
197            ## build ph_vocab/vocab_ph
198            ph_list = sw.ph_names
199            if len(ph_list) == 1:
200                vocab_ph[sw.vocab_name].add(ph_list[0])
201                ph_vocab[ph_list[0]].add(sw.vocab_name)
202            elif len(ph_list) > 1:
203                logger.error("not supported: one placeholder for one term (switch: %s)", sw_name)
204        self.ph_vocab = ph_vocab
205        self.vocab_ph = vocab_ph
206        self.vocab_shape = vocab_shape
207        ##
208
209    def build(self, input_data: List[InputData]|None, sw_info: Dict[str, SwitchTensor]) -> None:
210        if input_data is not None:
211            self._build_ph_values(input_data)
212        else:
213            self.ph_values = {}
214        if self.ph_values is None:
215            raise ValueError(f"ph_values in PlaceholderGraph must not be None in build.")
216        self._build_vocab_ph(self.ph_values, sw_info)
217    def __repr__(self) -> str:
218        return f"PlaceholderGraph(ph_vocab={self.ph_vocab}, vocab_ph={self.vocab_ph}, ph_values={self.ph_values}, vocab_shape={self.vocab_shape})"

400    def _build_vocab_var_type(self, ph_graph: PlaceholderGraph, vocab_set: VocabSet, embedding_generators: List[BaseEmbeddingGenerator], sw_info: Dict[str,SwitchTensor])-> Dict[str, VarType]:
401        """Builds a map: vocab_name => VarType (dataclass)."""
402        if ph_graph.vocab_shape is None:

501    def _embed_switch_without_ph(
502        self,
503        sw: SwitchTensor,
504        vocab_var: Dict[str, PlaceholderData | TorchTensorBase],
505        vocab_set: VocabSet,
506        embedding_generators: List[BaseEmbeddingGenerator],
507    ) -> PlaceholderData | TorchTensorBase:
508        """Assign a tensor to a switch that has no placeholder."""
509        vocab_name = sw.vocab_name
510        # the last matching embedding generator wins
511        matched = [eg for eg in embedding_generators if eg.is_embedding(vocab_name)]
512        if len(matched) > 0:
513            eg = matched[-1]
514            # dataset without placeholder
515            shape = list(list(sw.shape_set)[0])
516            var_ds: Optional[PlaceholderData] = None
