tprism.expl_graph

This module contains the computational explanation graph.

A computational graph is constructed by forward traversing the given explanation graph: consistency of templates (index symbols and shapes of tensors) is verified first, and then each path is evaluated as an einsum operation with optional operators.

 62class ExplNode:
 63    """
 64    A container for intermediate values and dryrun descriptors.
 65    - If dryrun=False: holds a concrete value (tensor/scalar) in `value`.
 66    - If dryrun=True: holds a descriptor dict in `desc`.
 67    """
 68    def __init__(self, kind: str, dryrun: bool, value: Any = None, desc: Dict[str, Any] = {}) -> None:
 69        self.kind = kind
 70        self.dryrun = dryrun
 71        self.value = value
 72        self.desc = {} if desc is None else desc
 73
 74    @staticmethod
 75    def from_value(kind: str, value: Any) -> "ExplNode":
 76        return ExplNode(kind=kind, dryrun=False, value=value)
 77
 78    @staticmethod
 79    def from_desc(kind: str, desc: Dict[str, Any]) -> "ExplNode":
 80        return ExplNode(kind=kind, dryrun=True, desc=desc)
 81
 82    def as_value(self) -> Any:
 83        if self.dryrun:
 84            raise RuntimeError("Attempted to access concrete value from a dryrun ExplNode")
 85        return self.value
 86
 87    def as_desc(self) -> Dict[str, Any]:
 88        if self.dryrun:
 89            return self.desc
 90        # Optional lightweight descriptor when not in dryrun (used rarely)
 91        return {"type": self.kind}
 92
 93    def add_path_scalar_descriptors(self, scalars: List["ExplNode"]) -> None:
 94        """
 95        Attach scalar path descriptors (for dryrun pretty output).
 96        """
 97        if self.dryrun:
 98            lst = []
 99            for s in scalars:
100                if isinstance(s, ExplNode):
101                    lst.append(s.as_desc())
102                else:
103                    lst.append(s)
104            self.desc["path_scalar"] = lst

139    def get_scalar_inside(self) -> Any:
140        """
141        Return a scalar usable in computations (non-dryrun only).
142        If inside is a list, convert to tensor and squeeze.
143        """
144        if self.dryrun:
145            # In dryrun, scalar usage is handled by caller constructing descriptors.

153    @classmethod
154    def merge_paths(
155        cls,
156        path_template_list: List[List[TensorIndexRef]],
157        path_inside: List["ExplNode"],
158        path_batch_flag: bool,
159        dryrun: bool,
160    ) -> "GoalInsideEntry":
161        """
162        Merge paths into a single GoalInsideEntry adhering to the previous behavior.
163        Supports lists of ExplNode.
164        """
165        def normalize_list(items: List["ExplNode"], is_dry: bool) -> List[Any]:
166            return [it.as_desc() if is_dry else it.as_value() for it in items]
167
168        if len(path_template_list) == 0:
169            template: List[TensorIndexRef] = []
170            inside_list = normalize_list(path_inside, dryrun)
171            inside: Any = inside_list if dryrun else torch.tensor(1)
172            # batch_flag must be False for this case (keep original behavior)
173            return cls(template, inside, False, dryrun)
174
175        if len(path_template_list) != 1:
176            logger.warning("missmatch indices: %s", path_template_list)
177
178        template = path_template_list[0]
179        if dryrun:
180            inside = normalize_list(path_inside, True)
181        else:
182            items = normalize_list(path_inside, False)
183            if len(template) == 0:
184                inside = items[0]
185            else:
186                inside = torch.sum(torch.stack(items), dim=0)
187        return cls(template, inside, path_batch_flag, dryrun)

190class ComputationalExplGraph(torch.nn.Module):
191    """ This class is a computational explanation graph for pytorch.
192
193    The method build_explanation_graph_template confirms consistency of
194    templates, index symbols and shape of tensors, in a given explanation
195    graph and computes a list of goal templates (goal means LHS of path).
196    The forward method converts the explanation graph into a computational
197    graph.
198
199Note:
200    A path in the explanation graph is represented as the following format:
201        ```
202        {
203            "sw_template": [],         a list of template: List[str]
204            "sw_inside": [],           a list of torch.tensor
205            "prob_sw_inside": [],      a list of torch.tensor (scalar)
206            "node_template": [],       a list of template: List[str]
207            "node_inside": [],         a list of torch.tensor
208            "node_scalar_inside": [],  a list of torch.tensor (scalar)
209        }
210        ```
211Note:
212    Each goal in the explanation graph is represented by GoalInsideEntry instead of a dict.
213    """
214    def __init__(
215        self,
216        graph: expl_pb2.ExplGraph,
217        tensor_provider: SwitchTensorProvider,
218        operator_loader: OperatorLoader,
219        cycle_embedding_generator: Optional[BaseEmbeddingGenerator] = None,
220    ) -> None:
221        super().__init__()
222        ## setting
223        self.operators: Dict[str, BaseOperator] = {}
224        self.operator_loader: Optional[OperatorLoader] = operator_loader
225        self.goal_template: List[Optional[GoalTemplate]] = []
226        self.cycle_node: List[int] = []
227        self.param_op: Optional[torch.nn.ModuleList] = None
228        self.graph: "expl_pb2.ExplGraph" = graph
229        self.loss: Dict[str, torch.Tensor] = {}
230        self.tensor_provider: SwitchTensorProvider = tensor_provider
231        self.cycle_embedding_generator: Optional[BaseEmbeddingGenerator] = cycle_embedding_generator
232        ###
233        self.build()

235    def _get_unique_symbol_list(self, seq: List[List[TensorIndexRef]]) -> List[List[TensorIndexRef]]:
236        seen: List[Tuple[str, ...]] = []
237        seen_: List[List[TensorIndexRef]] = []
238        for ts in seq:
239            symbols = tuple([t.symbol for t in ts])
240            if symbols not in seen:
241                seen.append(symbols)
242                seen_.append(ts)
243        return seen_

245    # [['i'], ['i','l', 'j'], ['j','k']] => ['l','k']
246    def _compute_output_template(self, template: List[List[TensorIndexRef]]) -> List[TensorIndexRef]:
247        # skip index
248        template_=[]
249        for ts in template:
250            l=[]
251            for t in ts:
252                if t.index_type!="index":  # Literal["symbol", "range", "index"]
253                    l.append(t.symbol)
254            template_.append(l)
255        # counting
256        counter = collections.Counter(chain.from_iterable(template_))
257        out_template_ = [k for k, cnt in counter.items() if cnt == 1 and k != "b"]
258        out_template = [TensorIndexRef("symbol", 0, -1, 1, e) for e in sorted(out_template_)]
259        return out_template

261    # [['i'], ['i','l', 'j'], ['j','k']] => ['l','k']
262    # [[3], [3, 4, 5], [5,6]] => [4,6]
263    def _compute_output_shape(self, out_template: List[TensorIndexRef], sw_node_template: List[List[TensorIndexRef]], sw_node_shape: List[Tuple[Optional[int], ...]]) -> Tuple[Optional[int],...]:
264        symbol_shape = {}
265        for template_list, shape_list in zip(sw_node_template, sw_node_shape):
266            extracted_shape = extract_tensor_shape(shape_list, template_list)
267
268            for t, s in zip(template_list, extracted_shape):
269                if t.symbol not in symbol_shape:
270                    symbol_shape[t.symbol] = s
271                elif symbol_shape[t.symbol] is None:
272                    symbol_shape[t.symbol] = s
273                else:
274                    assert symbol_shape[t.symbol] == s, (
275                        "index symbol mismatch:"
276                        + str(t)
277                        + ":"
278                        + str(symbol_shape[t.symbol])
279                        + "!="
280                        + str(s)
281                    )
282        out_shape = []
283        for el in out_template:
284            symbol = el.symbol
285            if symbol in symbol_shape:
286                out_shape.append(symbol_shape[symbol])
287            else:
288                out_shape.append(None)
289        return tuple(out_shape)

291    def _unify_shapes(self, path_shapes: List[Tuple[Optional[int],...]]) -> Tuple[Optional[int],...]:
292        """
293        This method is used to unify shapes for all paths
294        """
295        n = len(path_shapes)
296        if n == 0:
297            return ()
298        else:
299            m = len(path_shapes[0])
300            out_shape = []
301            for j in range(m):
302                dim = None
303                for i in range(n):
304                    if path_shapes[i][j] is None:
305                        pass
306                    elif dim is None:
307                        dim = path_shapes[i][j]
308                    else:
309                        assert path_shapes[i][j] == dim, "shape mismatching"
310                out_shape.append(dim)
311            return tuple(out_shape)

313    def _apply_operator_template(
314        self,
315        operator_loader: Optional[OperatorLoader],
316        op: expl_pb2.SwIns,
317        in_template: List[TensorIndexRef],
318        in_shape: Tuple[Optional[int], ...]
319    ) -> Tuple[List[TensorIndexRef], Tuple[Optional[int], ...]]:
320        """
321        Applies an operator template to the given input template and shape, restoring or creating the operator as needed.
322
323        Args:
324            operator_loader(OperatorLoader): An object responsible for loading operator classes by name.
325            op: An operator instance containing 'name' and 'values' attributes.
326            in_template: The input template to which the operator will be applied.
327            in_shape: The input shape to which the operator will be applied.
328
329        Returns:
330            tuple: A tuple (out_template, out_shape) where:
331                - out_template: The output template after applying the operator.
332                - out_shape: The output shape after applying the operator.
333
334        Side Effects:
335            Updates the self.operators dictionary with the operator instance if it does not already exist.
336
337        """
338        key: str = str(op.name) + "_" + str(op.values)
339        if key in self.operators:
340            op_obj = self.operators[key]
341        else:
342            if operator_loader is None:
343                raise ValueError("operator_loader must not be None when applying operator templates.")
344            cls = operator_loader.get_operator(op.name)
345            op_obj = cls(op.values)
346            self.operators[key] = op_obj
347        out_template = op_obj.get_output_template(in_template)
348        out_shape = op_obj.get_output_shape(in_shape)
349        return out_template, out_shape

351    def build_explanation_graph_template(
352        self,
353        graph: expl_pb2.ExplGraph,
354        tensor_provider: SwitchTensorProvider,
355        operator_loader: Optional[OperatorLoader] = None,
356        cycle_node: List[int] = [],
357    ) -> Tuple[List[Optional[GoalTemplate]], List[int]]:
358        """
359        Args:
360            graph(expl_pb2.ExplGraph): explanation graph object
361            tensor_provider (SwitchTensorProvider): tensor provider
362            operator_loader (OperatorLoader): operator loader
363            cycle_node (List[int]): a list of sorted_id of cycle nodes
364
365        Returns:
366        A tuple containing goal_template and cycle_node
367            - goal_template (List[GoalTemplate]): a list of goal templates
368            - cycle_node (List[int]): a list of cycle node (if given cycle_node, it is updated)
369        """
370        # checking template
371        goal_template: List[Optional[GoalTemplate]] = [None] * len(graph.goals)
372        for i in range(len(graph.goals)):
373            g = graph.goals[i]
374            path_template = []
375            path_shape = []
376            path_batch_flag = False
377            for path in g.paths:
378                ## build template and inside for switches in the path
379                sw_template: List[List[TensorIndexRef]] = []
380                sw_shape: List[Tuple[Optional[int],...]] = []
381                for sw in path.tensor_switches:
382                    ph = tensor_provider.get_placeholder_name(sw.name)
383                    sw_obj = tensor_provider.get_switch(sw.name)
384                    sw_index_list=[parse_tensor_index(el) for el in sw.values]
385                    if len(ph) > 0:
386                        sw_template.append([TensorIndexRef("symbol", 0, -1, 1, "b")] + sw_index_list)
387                        path_batch_flag = True
388                        sw_shape.append(tuple([None] + list(sw_obj.get_shape())))
389                    else:
390                        sw_template.append(sw_index_list)
391                        sw_shape.append(sw_obj.get_shape())
392                ## building template and inside for nodes in the path
393                node_template: List[List[TensorIndexRef]] = []
394                node_shape: List[Tuple[Optional[int],...]] = []
395                cycle_detected = False
396                for node in path.nodes:
397                    temp_goal = goal_template[node.sorted_id]
398                    if temp_goal is None:
399                        # cycle
400                        if node.sorted_id not in cycle_node:
401                            cycle_node.append(node.sorted_id)
402                        cycle_detected = True
403                        continue
404                    if len(temp_goal.template) > 0:
405                        if temp_goal.batch_flag:
406                            path_batch_flag = True
407                        node_shape.append(tuple(temp_goal.shape))
408                        node_template.append(temp_goal.template)
409                #if cycle_detected:
410                #    continue
411                sw_node_template = sw_template + node_template
412                sw_node_shape = sw_shape + node_shape
413
414                ##########
415                ops = [op.name for op in path.operators]
416                if "distribution" in ops:
417                    # distributino clause
418                    graph_logger.debug("=== distribution ===")
419                    graph_logger.debug("%s", sw_node_template)
420                    graph_logger.debug("%s", sw_node_shape)
421                    ##
422                    out_template = sw_node_template[0]
423                    out_shape = sw_node_shape[0]
424                    graph_logger.debug("%s", out_template)
425                    graph_logger.debug("%s", out_shape)
426                else:
427                    # constructing einsum operation using template and inside
428                    out_template = self._compute_output_template(sw_node_template)
429                    out_shape = self._compute_output_shape(
430                        out_template, sw_node_template, sw_node_shape
431                    )
432                    if len(sw_node_template) > 0:  # condition for einsum
433                        if path_batch_flag:
434                            out_template = [TensorIndexRef("symbol", 0, -1, 1, "b")] + out_template
435                    ## computing operaters
436                    for op in path.operators:
437                        out_template, out_shape=self._apply_operator_template(
438                            operator_loader,
439                            op,
440                            out_template,
441                            out_shape)
442                ##########
443                path_template.append(out_template)
444                path_shape.append(out_shape)
445                ##
446            ##
447            path_template_list = self._get_unique_symbol_list(path_template)
448            path_shape_ = self._unify_shapes(path_shape)
449            if len(path_template_list) == 0:
450                goal_template[i] = GoalTemplate(
451                    template=[],
452                    batch_flag=False,
453                    shape=path_shape_,
454                )
455            else:
456                if len(path_template_list) != 1:
457                    logger.warning("missmatch indices: %s", path_template_list)
458                goal_template[i] = GoalTemplate(
459                    template=path_template[0],
460                    batch_flag=path_batch_flag,
461                    shape=path_shape_,
462                )
463        ##
464        return goal_template, cycle_node

466    def build(self) -> None:
467
468        goal_template, cycle_node = self.build_explanation_graph_template(
469            self.graph, self.tensor_provider, self.operator_loader

471        self.goal_template = goal_template
472        self.cycle_node = cycle_node
473
474        ## setting parameterized operators
475        self.param_op = torch.nn.ModuleList()
476        for k, v in self.operators.items():
477            if isinstance(v, torch.nn.Module):
478                debug_logger("module").debug("Torch parameterized operator: %s", k)
479                self.param_op.append(cast(torch.nn.Module, v))
480        ##

482            self.register_parameter(name, param)

484    def _distribution_forward(
485        self,
486        name: str,
487        dist: str,
488        params: List[torch.Tensor],
489        param_template: List[List[TensorIndexRef]],
490        op: "expl_pb2.SwIns",
491    ) -> Tuple[torch.Tensor, List[TensorIndexRef]]:
492        if dist == "normal":
493            mean = params[0]
494            var = params[1]
495            scale = torch.sqrt(F.softplus(var))
496            q_z = torch.distributions.normal.Normal(mean, scale)
497            out_inside = q_z.rsample()
498            out_template = param_template[0]
499            p_z = torch.distributions.normal.Normal(
500                torch.zeros_like(mean), torch.ones_like(var)
501            )
502            loss_KL = torch.distributions.kl.kl_divergence(q_z, p_z)
503            self.loss[name] = loss_KL.sum()
504        else:
505            out_inside = params[0]
506            out_template = param_template[0]
507            logger.error("unknown distribution: %s", dist)
508        return out_inside, out_template

510    def make_einsum_args(
511        self,
512        template: List[List[TensorIndexRef]],
513        out_template: List[TensorIndexRef],
514        path_batch_flag: bool,
515    ) -> Tuple[str, List[TensorIndexRef]]:
516        """
517        Example
518        template: [["i","j"],["j","k"]]
519        out_template: ["i","k"]
520        => "ij,jk->ik", out_template
521        """
522        lhs = ",".join(map(lambda x: "".join([el.symbol for el in x]), template))
523        ## skip index
524        rhs_list=[]
525        for el in out_template:
526            if el.index_type!="index":  # Literal["symbol", "range", "index"]
527                rhs_list.append(el.symbol)
528        rhs = "".join(rhs_list)
529        if path_batch_flag:
530            rhs = "b" + rhs
531            out_template = [TensorIndexRef("symbol", 0, -1, 1, "b")] + out_template
532        einsum_eq = lhs + "->" + rhs
533        return einsum_eq, out_template

535    def make_einsum_args_sublist(
536        self,
537        template: List[List[TensorIndexRef]],
538        inputs: List[torch.Tensor],
539        out_template: List[TensorIndexRef],

542        """
543        Example
544        template: [["i","j"],["j","k"]]

548
549        symbol_set = set([e.symbol for sublist in template for e in sublist])

567    def _apply_operator(
568        self,
569        op: expl_pb2.SwIns,
570        operator_loader: Optional["OperatorLoader"],
571        out_node: ExplNode,
572        out_template: List[TensorIndexRef],
573        dryrun: bool,
574    ) -> Tuple[ExplNode, List[TensorIndexRef]]:
575        ## restore operator
576        key=str(op.name)+"_"+str(op.values)
577        op_obj:Optional[BaseOperator]=None
578        if key in self.operators:
579            op_obj=self.operators[key]
580        else:
581            assert True, "unknown operator "+key+" has been found in forward procedure"
582        if op_obj is None:
583            raise Exception("Unknown operator: ", key)
584        if dryrun:
585            desc = {
586                "type": "operator",
587                "name": op.name,
588                "path": out_node.as_desc(),
589            }
590            out_node = ExplNode.from_desc("operator", desc)
591        else:
592            out_val = op_obj.call(out_node.as_value())
593            out_node = ExplNode.from_value("operator", out_val)
594        out_template = op_obj.get_output_template(out_template)
595        return out_node, out_template

246    # [['i'], ['i','l', 'j'], ['j','k']] => ['l','k']
247    def _compute_output_template(self, template: List[List[TensorIndexRef]]) -> List[TensorIndexRef]:
248        # skip index
249        template_=[]
250        for ts in template:
251            l=[]
252            for t in ts:
253                if t.index_type!="index":  # Literal["symbol", "range", "index"]
254                    l.append(t.symbol)
255            template_.append(l)
256        # counting
257        counter = collections.Counter(chain.from_iterable(template_))
258        out_template_ = [k for k, cnt in counter.items() if cnt == 1 and k != "b"]
259        out_template = [TensorIndexRef("symbol", 0, -1, 1, e) for e in sorted(out_template_)]
260        return out_template

264    def _compute_output_shape(self, out_template: List[TensorIndexRef], sw_node_template: List[List[TensorIndexRef]], sw_node_shape: List[Tuple[Optional[int], ...]]) -> Tuple[Optional[int],...]:
265        symbol_shape = {}
266        for template_list, shape_list in zip(sw_node_template, sw_node_shape):
267            extracted_shape = extract_tensor_shape(shape_list, template_list)
268
269            for t, s in zip(template_list, extracted_shape):
270                if t.symbol not in symbol_shape:
271                    symbol_shape[t.symbol] = s
272                elif symbol_shape[t.symbol] is None:
273                    symbol_shape[t.symbol] = s
274                else:
275                    assert symbol_shape[t.symbol] == s, (
276                        "index symbol mismatch:"
277                        + str(t)
278                        + ":"
279                        + str(symbol_shape[t.symbol])
280                        + "!="
281                        + str(s)
282                    )
283        out_shape = []
284        for el in out_template:
285            symbol = el.symbol
286            if symbol in symbol_shape:
287                out_shape.append(symbol_shape[symbol])
288            else:
289                out_shape.append(None)
290        return tuple(out_shape)

292    def _unify_shapes(self, path_shapes: List[Tuple[Optional[int],...]]) -> Tuple[Optional[int],...]:
293        """
294        This method is used to unify shapes for all paths
295        """
296        n = len(path_shapes)
297        if n == 0:
298            return ()
299        else:
300            m = len(path_shapes[0])
301            out_shape = []
302            for j in range(m):
303                dim = None
304                for i in range(n):
305                    if path_shapes[i][j] is None:
306                        pass
307                    elif dim is None:
308                        dim = path_shapes[i][j]
309                    else:
310                        assert path_shapes[i][j] == dim, "shape mismatching"
311                out_shape.append(dim)
312            return tuple(out_shape)

314    def _apply_operator_template(
315        self,
316        operator_loader: Optional[OperatorLoader],
317        op: expl_pb2.SwIns,
318        in_template: List[TensorIndexRef],
319        in_shape: Tuple[Optional[int], ...]
320    ) -> Tuple[List[TensorIndexRef], Tuple[Optional[int], ...]]:
321        """
322        Applies an operator template to the given input template and shape, restoring or creating the operator as needed.
323
324        Args:
325            operator_loader(OperatorLoader): An object responsible for loading operator classes by name.
326            op: An operator instance containing 'name' and 'values' attributes.
327            in_template: The input template to which the operator will be applied.
328            in_shape: The input shape to which the operator will be applied.
329
330        Returns:
331            tuple: A tuple (out_template, out_shape) where:
332                - out_template: The output template after applying the operator.
333                - out_shape: The output shape after applying the operator.
334
335        Side Effects:
336            Updates the self.operators dictionary with the operator instance if it does not already exist.
337
338        """
339        key: str = str(op.name) + "_" + str(op.values)
340        if key in self.operators:
341            op_obj = self.operators[key]
342        else:
343            if operator_loader is None:
344                raise ValueError("operator_loader must not be None when applying operator templates.")
345            cls = operator_loader.get_operator(op.name)
346            op_obj = cls(op.values)
347            self.operators[key] = op_obj
348        out_template = op_obj.get_output_template(in_template)
349        out_shape = op_obj.get_output_shape(in_shape)
350        return out_template, out_shape

352    def build_explanation_graph_template(
353        self,
354        graph: expl_pb2.ExplGraph,
355        tensor_provider: SwitchTensorProvider,
356        operator_loader: Optional[OperatorLoader] = None,
357        cycle_node: List[int] = [],
358    ) -> Tuple[List[Optional[GoalTemplate]], List[int]]:
359        """
360        Args:
361            graph(expl_pb2.ExplGraph): explanation graph object
362            tensor_provider (SwitchTensorProvider): tensor provider
363            operator_loader (OperatorLoader): operator loader
364            cycle_node (List[int]): a list of sorted_id of cycle nodes
365
366        Returns:
367        A tuple containing goal_template and cycle_node
368            - goal_template (List[GoalTemplate]): a list of goal templates
369            - cycle_node (List[int]): a list of cycle node (if given cycle_node, it is updated)
370        """
371        # checking template
372        goal_template: List[Optional[GoalTemplate]] = [None] * len(graph.goals)
373        for i in range(len(graph.goals)):
374            g = graph.goals[i]
375            path_template = []
376            path_shape = []
377            path_batch_flag = False
378            for path in g.paths:
379                ## build template and inside for switches in the path
380                sw_template: List[List[TensorIndexRef]] = []
381                sw_shape: List[Tuple[Optional[int],...]] = []
382                for sw in path.tensor_switches:
383                    ph = tensor_provider.get_placeholder_name(sw.name)
384                    sw_obj = tensor_provider.get_switch(sw.name)
385                    sw_index_list=[parse_tensor_index(el) for el in sw.values]
386                    if len(ph) > 0:
387                        sw_template.append([TensorIndexRef("symbol", 0, -1, 1, "b")] + sw_index_list)
388                        path_batch_flag = True
389                        sw_shape.append(tuple([None] + list(sw_obj.get_shape())))
390                    else:
391                        sw_template.append(sw_index_list)
392                        sw_shape.append(sw_obj.get_shape())
393                ## building template and inside for nodes in the path
394                node_template: List[List[TensorIndexRef]] = []
395                node_shape: List[Tuple[Optional[int],...]] = []
396                cycle_detected = False
397                for node in path.nodes:
398                    temp_goal = goal_template[node.sorted_id]
399                    if temp_goal is None:
400                        # cycle
401                        if node.sorted_id not in cycle_node:
402                            cycle_node.append(node.sorted_id)
403                        cycle_detected = True
404                        continue
405                    if len(temp_goal.template) > 0:
406                        if temp_goal.batch_flag:
407                            path_batch_flag = True
408                        node_shape.append(tuple(temp_goal.shape))
409                        node_template.append(temp_goal.template)
410                #if cycle_detected:
411                #    continue
412                sw_node_template = sw_template + node_template
413                sw_node_shape = sw_shape + node_shape
414
415                ##########
416                ops = [op.name for op in path.operators]
417                if "distribution" in ops:
418                    # distributino clause
419                    graph_logger.debug("=== distribution ===")
420                    graph_logger.debug("%s", sw_node_template)
421                    graph_logger.debug("%s", sw_node_shape)
422                    ##
423                    out_template = sw_node_template[0]
424                    out_shape = sw_node_shape[0]
425                    graph_logger.debug("%s", out_template)
426                    graph_logger.debug("%s", out_shape)
427                else:
428                    # constructing einsum operation using template and inside
429                    out_template = self._compute_output_template(sw_node_template)
430                    out_shape = self._compute_output_shape(
431                        out_template, sw_node_template, sw_node_shape
432                    )
433                    if len(sw_node_template) > 0:  # condition for einsum
434                        if path_batch_flag:
435                            out_template = [TensorIndexRef("symbol", 0, -1, 1, "b")] + out_template
436                    ## computing operaters
437                    for op in path.operators:
438                        out_template, out_shape=self._apply_operator_template(
439                            operator_loader,
440                            op,
441                            out_template,
442                            out_shape)
443                ##########
444                path_template.append(out_template)
445                path_shape.append(out_shape)
446                ##
447            ##
448            path_template_list = self._get_unique_symbol_list(path_template)
449            path_shape_ = self._unify_shapes(path_shape)
450            if len(path_template_list) == 0:
451                goal_template[i] = GoalTemplate(
452                    template=[],
453                    batch_flag=False,
454                    shape=path_shape_,
455                )
456            else:
457                if len(path_template_list) != 1:
458                    logger.warning("missmatch indices: %s", path_template_list)
459                goal_template[i] = GoalTemplate(
460                    template=path_template[0],
461                    batch_flag=path_batch_flag,
462                    shape=path_shape_,
463                )
464        ##
465        return goal_template, cycle_node

467    def build(self) -> None:
468
469        goal_template, cycle_node = self.build_explanation_graph_template(
470            self.graph, self.tensor_provider, self.operator_loader
471        )
472        self.goal_template = goal_template
473        self.cycle_node = cycle_node
474
475        ## setting parameterized operators
476        self.param_op = torch.nn.ModuleList()
477        for k, v in self.operators.items():
478            if isinstance(v, torch.nn.Module):
479                debug_logger("module").debug("Torch parameterized operator: %s", k)
480                self.param_op.append(cast(torch.nn.Module, v))
481        ##
482        for name, (param, tensor_type) in self.tensor_provider.params.items():
483            self.register_parameter(name, param)

485    def _distribution_forward(
486        self,
487        name: str,
488        dist: str,
489        params: List[torch.Tensor],
490        param_template: List[List[TensorIndexRef]],
491        op: "expl_pb2.SwIns",
492    ) -> Tuple[torch.Tensor, List[TensorIndexRef]]:
493        if dist == "normal":
494            mean = params[0]
495            var = params[1]
496            scale = torch.sqrt(F.softplus(var))
497            q_z = torch.distributions.normal.Normal(mean, scale)
498            out_inside = q_z.rsample()
499            out_template = param_template[0]
500            p_z = torch.distributions.normal.Normal(
501                torch.zeros_like(mean), torch.ones_like(var)
502            )
503            loss_KL = torch.distributions.kl.kl_divergence(q_z, p_z)
504            self.loss[name] = loss_KL.sum()
505        else:
506            out_inside = params[0]
507            out_template = param_template[0]
508            logger.error("unknown distribution: %s", dist)
509        return out_inside, out_template

511    def make_einsum_args(
512        self,
513        template: List[List[TensorIndexRef]],
514        out_template: List[TensorIndexRef],
515        path_batch_flag: bool,
516    ) -> Tuple[str, List[TensorIndexRef]]:
517        """
518        Example
519        template: [["i","j"],["j","k"]]
520        out_template: ["i","k"]
521        => "ij,jk->ik", out_template
522        """
523        lhs = ",".join(map(lambda x: "".join([el.symbol for el in x]), template))
524        ## skip index
525        rhs_list=[]
526        for el in out_template:
527            if el.index_type!="index":  # Literal["symbol", "range", "index"]
528                rhs_list.append(el.symbol)
529        rhs = "".join(rhs_list)
530        if path_batch_flag:
531            rhs = "b" + rhs
532            out_template = [TensorIndexRef("symbol", 0, -1, 1, "b")] + out_template
533        einsum_eq = lhs + "->" + rhs
534        return einsum_eq, out_template

536    def make_einsum_args_sublist(
537        self,
538        template: List[List[TensorIndexRef]],
539        inputs: List[torch.Tensor],
540        out_template: List[TensorIndexRef],
541        path_batch_flag: bool,
542    ) -> Tuple[List[torch.Tensor| List[int]], List[TensorIndexRef]]:
543        """
544        Example
545        template: [["i","j"],["j","k"]]
546        out_template: ["i","k"]
547        => [inputs[0], [0,1], inputs[1], [1,2], [0,2]], out_template
548        """
549
550        symbol_set = set([e.symbol for sublist in template for e in sublist])
551        mapping={s:i for i,s in enumerate(symbol_set)}
552        if path_batch_flag:
553            out_template = [TensorIndexRef("symbol", 0, -1, 1, "b")] + out_template
554        sublistform_args: List[torch.Tensor| List[int]] = []
555        for v,input_x in zip(template,inputs):
556            ## skip index
557            l=[]
558            for el in v:
559                if el.index_type!="index":  # Literal["symbol", "range", "index"]
560                    l.append(mapping[el.symbol])
561            ##
562            slice_x, index_tuple, out_symbols=extract_tensor(input_x, v)
563            sublistform_args.append(slice_x)
564            sublistform_args.append(l)
565        sublistform_args.append([mapping[el.symbol] for el in out_template])
566        return sublistform_args, out_template

568    def _apply_operator(
569        self,
570        op: expl_pb2.SwIns,
571        operator_loader: Optional["OperatorLoader"],
572        out_node: ExplNode,
573        out_template: List[TensorIndexRef],
574        dryrun: bool,
575    ) -> Tuple[ExplNode, List[TensorIndexRef]]:
576        ## restore operator
577        key=str(op.name)+"_"+str(op.values)
578        op_obj:Optional[BaseOperator]=None
579        if key in self.operators:
580            op_obj=self.operators[key]
581        else:
582            assert True, "unknown operator "+key+" has been found in forward procedure"
583        if op_obj is None:
584            raise Exception("Unknown operator: ", key)
585        if dryrun:
586            desc = {
587                "type": "operator",
588                "name": op.name,
589                "path": out_node.as_desc(),
590            }
591            out_node = ExplNode.from_desc("operator", desc)
592        else:
593            out_val = op_obj.call(out_node.as_value())
594            out_node = ExplNode.from_value("operator", out_val)
595        out_template = op_obj.get_output_template(out_template)
596        return out_node, out_template

598    def forward_path_node(
599        self,
600        path: expl_pb2.ExplGraphPath,
601        goal_inside: List[Optional[GoalInsideEntry]],
602        dryrun: bool = False,
603    ) -> Tuple[List[List[TensorIndexRef]], List[ExplNode], List[ExplNode], bool]:
604        goal_template = self.goal_template
605        cycle_embedding_generator = self.cycle_embedding_generator
606        cycle_node=self.cycle_node
607        node_template = []
608        node_inside = []
609        node_scalar_inside = []
610        path_batch_flag=False
611        for node in path.nodes:
612            temp_goal = goal_inside[node.sorted_id]
613            if node.sorted_id in cycle_node:
614                name = node.goal.name
615                gt=goal_template[node.sorted_id]
616                if  gt is None:
617                    raise Exception("goal_template[", node.sorted_id,"] is None in cycle node")
618                template = gt["template"]
619                shape = cast(Tuple[int], gt["shape"])
620                if dryrun:
621                    args = node.goal.args
622                    temp_goal_inside = ExplNode.from_desc("goal", {
623                        "type":"goal",
624                        "from":"cycle_embedding_generator",
625                        "name": name,
626                        "args": args,
627                        "id": node.sorted_id,
628                        "shape":shape})
629                else:
630                    if cycle_embedding_generator is None:
631                        raise Exception("cycle_embedding_generator is None in cycle node")
632                    else:
633                        v = cycle_embedding_generator.forward(name, shape, node.sorted_id)
634                        temp_goal_inside = ExplNode.from_value("goal", v)
635                temp_goal_template = template
636                node_inside.append(temp_goal_inside)
637                node_template.append(temp_goal_template)
638            elif temp_goal is None:
639                logger.error(
640                    "cycle node is detected: sorted_id=%s node=%s temp_goal=%s",
641                    node.sorted_id,
642                    node,
643                    goal_inside[node.sorted_id],
644                )
645                quit()
646            elif len(temp_goal.template) > 0:
647                # tensor subgoal
648                if dryrun:
649                    name = node.goal.name
650                    args = node.goal.args
651                    temp_goal_inside = ExplNode.from_desc("goal", {
652                        "type":"goal",
653                        "from":"goal",
654                        "name": name,
655                        "args": args,
656                        "id": node.sorted_id,})
657                else:
658                    temp_goal_inside = ExplNode.from_value("goal", temp_goal.inside)
659                temp_goal_template = temp_goal.template
660                if temp_goal.batch_flag:
661                    path_batch_flag = True
662                node_inside.append(temp_goal_inside)
663                node_template.append(temp_goal_template)
664            else:  # scalar subgoal
665                if dryrun:
666                    name = node.goal.name
667                    args = node.goal.args
668                    temp_goal_inside = ExplNode.from_desc("goal", {
669                        "type":"goal",
670                        "from":"goal",
671                        "name": name,
672                        "args": args,
673                        "id": node.sorted_id,})
674                    node_scalar_inside.append(temp_goal_inside)
675                else:
676                    node_scalar_inside.append(ExplNode.from_value("goal_scalar", temp_goal.get_scalar_inside()))
677        return node_template, node_inside, node_scalar_inside, path_batch_flag

718    def forward_path_op(
719        self,
720        path_name: str,
721        ops: Dict[str, "expl_pb2.SwIns"],
722        operator_loader: Optional[OperatorLoader],
723        sw_node_template: List[List[TensorIndexRef]],
724        sw_node_inside: List[ExplNode],
725        node_scalar_inside: List[ExplNode],
726        prob_sw_inside: List[ExplNode],
727        path_batch_flag: bool,
728        dryrun: bool = False,
729    ) -> Tuple[ExplNode, List[TensorIndexRef]]:
730
731        if "distribution" in ops:
732            op = ops["distribution"]
733            dist = op.values[0]
734            if dryrun:
735                desc = {
736                    "type": "distribution",
737                    "name": path_name,
738                    "dist_type": op,
739                    "path": [x.as_desc() if isinstance(x, ExplNode) else x for x in sw_node_inside],
740                }
741                out_node = ExplNode.from_desc("distribution", desc)
742                out_template= sw_node_template[0]  # TODO: refine if needed
743            else:
744                params = [x.as_value() if isinstance(x, ExplNode) else x for x in sw_node_inside]
745                out_val, out_template_temp = self._distribution_forward(
746                    path_name,
747                    dist,
748                    params=params,
749                    param_template=sw_node_template,
750                    op=op,
751                )
752                out_template = out_template_temp
753                out_node = ExplNode.from_value("distribution", out_val)
754
755        else:  # einsum operator
756            path_v = sorted(zip(sw_node_template, sw_node_inside), key=lambda x: x[0])
757            template = [x[0] for x in path_v]
758            inside_nodes = [x[1] for x in path_v]
759            out_template = self._compute_output_template(template)  # out_template is List[str]
760            if len(template) > 0:
761                if dryrun:
762                    #print(">in>",template)
763                    #print(">out>",out_template)
764                    einsum_eq, out_template_einsum = self.make_einsum_args(
765                        template, out_template, path_batch_flag)
766                    desc = {
767                        "type":"einsum",
768                        "name":"torch.einsum",
769                        "einsum_eq":einsum_eq,
770                        "path":[n.as_desc() if isinstance(n, ExplNode) else n for n in inside_nodes],
771                    }
772                    out_node = ExplNode.from_desc("einsum", desc)
773                    out_template = out_template_einsum
774                else:
775                    inside_vals = [n.as_value() if isinstance(n, ExplNode) else n for n in inside_nodes]
776                    einsum_args, out_template_temp = self.make_einsum_args_sublist(template, inside_vals, out_template, path_batch_flag)
777                    try:
778                        out_val = torch.einsum(*einsum_args)
779                    except Exception as e:
780                        logger.error("einsum failed for path: %s", path_name)
781                        logger.error("in  > %s", template)
782                        logger.error("out_org  > %s", out_template)
783                        logger.error("out_list > %s", out_template_temp)
784                        logger.error("inside_vals: %s", inside_vals)
785                        logger.error("einsum_args: %s", einsum_args)
786                        raise e
787                    out_node = ExplNode.from_value("einsum", out_val)
788                    out_template = out_template_temp
